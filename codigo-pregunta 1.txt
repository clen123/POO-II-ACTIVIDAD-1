#include <iostream>
using namespace std;

class Laptop{
private:
    int precio;
    string marca;
    string componentes;

public:
    Laptop(int, string, string);
    void encender();
    void apagar();
    void mostrar();
};

Laptop::Laptop(int _precio, string _marca, string _componentes){
    precio = _precio;
    marca = _marca;
    componentes = _componentes;
}

void Laptop::encender(){
    cout << "La laptop se esta encendiendo" << endl;
}

void Laptop::apagar(){
    cout << "La laptop se esta apagando" << endl;
}

void Laptop::mostrar(){
    cout << "Marca: " << marca << endl;
    cout << "Precio: " << precio << endl;
    cout << "Componentes: " << componentes << endl;
}

int main(){

    Laptop laptop(2500, "ASUS", "RAM, SSD y procesador");

    laptop.encender();
    laptop.mostrar();
    laptop.apagar();

    return 0;
}
